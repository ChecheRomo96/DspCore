224 bool processSample(const SampleType input, SampleType& output) noexcept {
225 if (numerator_.empty()) {
226 output = input;
227 return true;
228 }
229
230 const StateType stateInput = static_cast<StateType>(input);
231 if (!DspCore::Detail::Signal::Finite(stateInput)) {
232 return false;
233 }
234
235 StateType sample = static_cast<StateType>(numerator_[0]) * stateInput;
236 for (cpstd::size_t index = 1U; index < numerator_.size(); ++index) {
237 sample += static_cast<StateType>(numerator_[index]) * inputHistory_[index - 1U];
238 }
239 for (cpstd::size_t index = 1U; index < denominator_.size(); ++index) {
240 sample -= static_cast<StateType>(denominator_[index]) * outputHistory_[index - 1U];
241 }
242 if (!DspCore::Detail::Signal::Finite(sample)) {
243 return false;
244 }
245 SampleType converted{};
246 if (!DspCore::Detail::Signal::Convert(sample, converted)) {
247 return false;
248 }
249
250 for (cpstd::size_t index = inputHistory_.size(); index > 1U; --index) {
251 inputHistory_[index - 1U] = inputHistory_[index - 2U];
252 }
253 for (cpstd::size_t index = outputHistory_.size(); index > 1U; --index) {
254 outputHistory_[index - 1U] = outputHistory_[index - 2U];
255 }
256 if (!inputHistory_.empty()) {
257 inputHistory_[0] = stateInput;
258 }
259 if (!outputHistory_.empty()) {
260 outputHistory_[0] = sample;
261 }
262 output = converted;
263 return true;
264 }
