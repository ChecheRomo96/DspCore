149inline bool HasParameter(const Type type, const Parameter parameter) noexcept {
150 const auto count = ParameterCount(type);
151 for (cpstd::size_t index = 0U; index < count; ++index) {
152 ParameterInfo info;
153 if (ParameterAt(type, index, info) && info.parameter == parameter) {
154 return true;
155 }
156 }
157 return false;

173template <typename CoefficientType>
174inline bool ValidFrequency(const CoefficientType sampleRate, const CoefficientType cutoff,
175 const CoefficientType quality) noexcept {
176 static_assert(cpstd::is_same<CoefficientType, float>::value || cpstd::is_same<CoefficientType, double>::value,
177 "Biquad coefficient type must be float or double");
178 return DspCore::Detail::Signal::Finite(sampleRate) && DspCore::Detail::Signal::Finite(cutoff) &&
179 DspCore::Detail::Signal::Finite(quality) &&
180 sampleRate > CoefficientType{} && cutoff > CoefficientType{} &&
181 cutoff < sampleRate / CoefficientType{2} && quality > CoefficientType{};
182}

184template <typename CoefficientType>
185inline void Terms(const CoefficientType sampleRate, const CoefficientType cutoff,
186 const CoefficientType quality, CoefficientType& cosine, CoefficientType& alpha) noexcept {
187 const auto omega = CoefficientType{2} * DspCore::Detail::Signal::Pi<CoefficientType>() * cutoff / sampleRate;
188 cosine = DspCore::Detail::Signal::Cos(omega);
189 alpha = DspCore::Detail::Signal::Sin(omega) / (CoefficientType{2} * quality);
190}

198template <typename CoefficientType>
199inline Coefficients<CoefficientType> LowPass(const CoefficientType sampleRate, const CoefficientType cutoff,
200 const CoefficientType quality) noexcept {
201 if (!Detail::ValidFrequency(sampleRate, cutoff, quality)) return Detail::Invalid<CoefficientType>();
202 CoefficientType cosine{}, alpha{};
203 Detail::Terms(sampleRate, cutoff, quality, cosine, alpha);
204 return Detail::Make((CoefficientType{1} - cosine) / CoefficientType{2}, CoefficientType{1} - cosine,
205 (CoefficientType{1} - cosine) / CoefficientType{2}, CoefficientType{1} + alpha,
206 -CoefficientType{2} * cosine, CoefficientType{1} - alpha);
207}

256template <typename CoefficientType>
257inline Coefficients<CoefficientType> Peak(const CoefficientType sampleRate, const CoefficientType cutoff,
258 const CoefficientType quality, const CoefficientType gainDecibels) noexcept {
259 if (!Detail::ValidFrequency(sampleRate, cutoff, quality) || !DspCore::Detail::Signal::Finite(gainDecibels)) {
260 return Detail::Invalid<CoefficientType>();
261 }
262 CoefficientType cosine{}, alpha{};
263 Detail::Terms(sampleRate, cutoff, quality, cosine, alpha);
264 const auto amplitude = DspCore::Detail::Signal::Pow(CoefficientType{10}, gainDecibels / CoefficientType{40});
265 return Detail::Make(CoefficientType{1} + alpha * amplitude, -CoefficientType{2} * cosine,
266 CoefficientType{1} - alpha * amplitude, CoefficientType{1} + alpha / amplitude,
267 -CoefficientType{2} * cosine, CoefficientType{1} - alpha / amplitude);
268}

271template <typename CoefficientType>
272inline Coefficients<CoefficientType> LowShelf(const CoefficientType sampleRate, const CoefficientType cutoff,
273 const CoefficientType slope, const CoefficientType gainDecibels) noexcept {
274 if (!Detail::ValidFrequency(sampleRate, cutoff, slope) || slope > CoefficientType{1} ||
275 !DspCore::Detail::Signal::Finite(gainDecibels)) {
276 return Detail::Invalid<CoefficientType>();
277 }
278 const auto omega = CoefficientType{2} * DspCore::Detail::Signal::Pi<CoefficientType>() * cutoff / sampleRate;
279 const auto cosine = DspCore::Detail::Signal::Cos(omega);
280 const auto sine = DspCore::Detail::Signal::Sin(omega);
281 const auto amplitude = DspCore::Detail::Signal::Pow(CoefficientType{10}, gainDecibels / CoefficientType{40});
282 const auto alpha = sine / CoefficientType{2} * DspCore::Detail::Signal::Sqrt(
283 (amplitude + CoefficientType{1} / amplitude) *
284 (CoefficientType{1} / slope - CoefficientType{1}) + CoefficientType{2});
285 const auto beta = CoefficientType{2} * DspCore::Detail::Signal::Sqrt(amplitude) * alpha;
286 const auto plus = amplitude + CoefficientType{1};
287 const auto minus = amplitude - CoefficientType{1};
288 return Detail::Make(amplitude * (plus - minus * cosine + beta),
289 CoefficientType{2} * amplitude * (minus - plus * cosine),
290 amplitude * (plus - minus * cosine - beta), plus + minus * cosine + beta,
291 -CoefficientType{2} * (minus + plus * cosine), plus + minus * cosine - beta);
292}

297 const CoefficientType slope, const CoefficientType gainDecibels) noexcept {

299 !DspCore::Detail::Signal::Finite(gainDecibels)) {
300 return Detail::Invalid<CoefficientType>();
301 }

303 const auto cosine = DspCore::Detail::Signal::Cos(omega);
304 const auto sine = DspCore::Detail::Signal::Sin(omega);
305 const auto amplitude = DspCore::Detail::Signal::Pow(CoefficientType{10}, gainDecibels / CoefficientType{40});
306 const auto alpha = sine / CoefficientType{2} * DspCore::Detail::Signal::Sqrt(
307 (amplitude + CoefficientType{1} / amplitude) *
308 (CoefficientType{1} / slope - CoefficientType{1}) + CoefficientType{2});
309 const auto beta = CoefficientType{2} * DspCore::Detail::Signal::Sqrt(amplitude) * alpha;
310 const auto plus = amplitude + CoefficientType{1};
311 const auto minus = amplitude - CoefficientType{1};
312 return Detail::Make(amplitude * (plus + minus * cosine + beta),
313 -CoefficientType{2} * amplitude * (minus + plus * cosine),
314 amplitude * (plus + minus * cosine - beta), plus - minus * cosine + beta,
315 CoefficientType{2} * (minus - plus * cosine), plus - minus * cosine - beta);
316}

372template <typename CoefficientSet>
373inline bool IsStable(const CoefficientSet& coefficients) noexcept {
374 if (coefficients.denominator.size() != 3U) return false;
375 const auto a0 = coefficients.denominator[0];
376 if (a0 == decltype(a0){}) return false;
377 const auto a1 = coefficients.denominator[1] / a0;
378 const auto a2 = coefficients.denominator[2] / a0;
379 const auto one = decltype(a0){1};
380 return DspCore::Detail::Signal::Finite(a1) && DspCore::Detail::Signal::Finite(a2) && a2 > -one && a2 < one &&
381 a1 > -(one + a2) && a1 < one + a2;
382}
