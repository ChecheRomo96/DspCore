set(DSPCORE_FILTERS_ROOT_HEADERS
    "${CMAKE_CURRENT_LIST_DIR}/Biquad.h"
    "${CMAKE_CURRENT_LIST_DIR}/FIR.h"
    "${CMAKE_CURRENT_LIST_DIR}/IIR.h"
    "${CMAKE_CURRENT_LIST_DIR}/MIMO.h"
    "${CMAKE_CURRENT_LIST_DIR}/Multirate.h"
    "${CMAKE_CURRENT_LIST_DIR}/SOS.h")
set(DSPCORE_FILTERS_SINGLE_RATE_HEADERS
    "${CMAKE_CURRENT_LIST_DIR}/SingleRate/FIR.h"
    "${CMAKE_CURRENT_LIST_DIR}/SingleRate/IIR.h")
set(DSPCORE_FILTERS_HEADERS
    ${DSPCORE_FILTERS_ROOT_HEADERS}
    ${DSPCORE_FILTERS_SINGLE_RATE_HEADERS})
set(DSPCORE_FILTERS_DOCUMENTATION
    "${CMAKE_CURRENT_LIST_DIR}/Filters.dox"
    "${CMAKE_CURRENT_LIST_DIR}/SingleRate/SingleRate.dox")
target_sources(DspCore PRIVATE ${DSPCORE_FILTERS_HEADERS})
dspcore_stage_headers(${DSPCORE_FILTERS_HEADERS})
dspcore_add_dox(${DSPCORE_FILTERS_HEADERS} ${DSPCORE_FILTERS_DOCUMENTATION})
install(FILES ${DSPCORE_FILTERS_ROOT_HEADERS}
    DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}/DspCore/Filters" CONFIGURATIONS Release)
install(FILES ${DSPCORE_FILTERS_SINGLE_RATE_HEADERS}
    DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}/DspCore/Filters/SingleRate"
    CONFIGURATIONS Release)
