13template<typename Real, typename T>
14Real Sample(T value) noexcept {
15 const long double wide = static_cast<long double>(value);
16 const long double limit = static_cast<long double>(cpstd::numeric_limits<Real>::max());
17 if (wide != wide) return cpstd::numeric_limits<Real>::quiet_NaN();
18 if (wide > limit) return cpstd::numeric_limits<Real>::infinity();
19 if (wide < -limit) return -cpstd::numeric_limits<Real>::infinity();
20 return static_cast<Real>(wide);
21}
22template<typename Real>
23bool Axis(size_t count, Real coordinate, Boundary boundary,
24 size_t& first, size_t& second, Real& fraction) noexcept {
25 if (count == 0 || !Finite(coordinate)) return false;
26 if (boundary != Boundary::Clamp && boundary != Boundary::Wrap && boundary != Boundary::Reject)
27 return false;
28 if (boundary == Boundary::Reject && (coordinate < Real(0) || coordinate > Real(1))) return false;
29 if (count == 1) { first = second = 0; fraction = Real(0); return true; }
30 Real normalized = coordinate;
31 const bool wrap = boundary == Boundary::Wrap;
32 if (wrap) {
33 normalized = static_cast<Real>(fmod(static_cast<double>(coordinate), 1.0));
34 if (normalized < Real(0)) normalized += Real(1);
35 // Negative sub-ulp offsets can round to one after addition.
36 if (normalized >= Real(1)) normalized = Real(0);
37 } else {
38 if (normalized <= Real(0)) { first = second = 0; fraction = Real(0); return true; }
39 if (normalized >= Real(1)) { first = second = count - 1; fraction = Real(0); return true; }
40 }
41 // Require the axis extent to fit the calculation type's exact-integer range.
42 // This also prevents rounding an index beyond size_t before its cast.
43 const Real exactLimit = static_cast<Real>(ldexp(1.0, cpstd::numeric_limits<Real>::digits));
44 const Real extent = static_cast<Real>(count);
45 if (extent > exactLimit || extent >= static_cast<Real>(static_cast<size_t>(-1))) return false;
46 if (static_cast<size_t>(extent) != count) return false;
47 const Real scale = static_cast<Real>(wrap ? count : count - 1);
48 const Real position = normalized * scale;
49 if (position >= scale) {
50 first = count - 1;
51 second = wrap ? 0 : first;
52 fraction = wrap ? Real(1) : Real(0);
53 return true;
54 }
55 first = static_cast<size_t>(position);
56 second = first + 1;
57 if (second >= count) second = wrap ? 0 : first;
58 fraction = position - static_cast<Real>(first);
59 return true;
60}
