set(DSPCORE_GENERATORS_HEADERS
    "${CMAKE_CURRENT_LIST_DIR}/Periodic.h"
    "${CMAKE_CURRENT_LIST_DIR}/Shaping.h"
    "${CMAKE_CURRENT_LIST_DIR}/Envelopes.h"
    "${CMAKE_CURRENT_LIST_DIR}/Noise.h"
    "${CMAKE_CURRENT_LIST_DIR}/Windows.h")
if(DSPCORE_LUT AND DSPCORE_INTERPOLATION)
    list(APPEND DSPCORE_GENERATORS_HEADERS "${CMAKE_CURRENT_LIST_DIR}/LUTSegment.h")
endif()
set(DSPCORE_GENERATORS_DETAIL "${CMAKE_CURRENT_LIST_DIR}/Detail/Common.h")
target_sources(DspCore PRIVATE ${DSPCORE_GENERATORS_HEADERS} ${DSPCORE_GENERATORS_DETAIL})
dspcore_stage_headers(${DSPCORE_GENERATORS_HEADERS} ${DSPCORE_GENERATORS_DETAIL})
dspcore_add_dox(${DSPCORE_GENERATORS_HEADERS} "${CMAKE_CURRENT_LIST_DIR}/Generators.dox")
install(FILES ${DSPCORE_GENERATORS_HEADERS}
    DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}/DspCore/Generators" CONFIGURATIONS Release)
install(FILES ${DSPCORE_GENERATORS_DETAIL}
    DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}/DspCore/Generators/Detail" CONFIGURATIONS Release)
