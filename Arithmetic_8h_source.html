28template<typename Real>
29Real NormalizedLinear(Real value0, Real value1, Real t) noexcept {
30 static_assert(cpstd::is_same<Real, float>::value || cpstd::is_same<Real, double>::value,
31 "Interpolation arithmetic must explicitly use float or double");
32 if (t == Real(0)) return value0;
33 if (t == Real(1)) return value1;
34 // Weighted form avoids overflowing value1-value0 for opposite-sign finite endpoints.
35 if ((value0 <= Real(0) && value1 >= Real(0)) || (value0 >= Real(0) && value1 <= Real(0)))
36 return (Real(1) - t) * value0 + t * value1;
37 return value0 + t * (value1 - value0);
38}
