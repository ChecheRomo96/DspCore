Inspect a generic numerator/denominator IIR response to a step pulse.

Goal

Configure a generic IIR with four numerator and two denominator coefficients, including an a0 normalization from two, then observe its stateful response to a pulse that is high from 10 ms through 40 ms.

Run and render

./build/macos_arm64/bin/Debug/DspCore_Filters_IIR_d --output ./output
./build/macos_arm64/bin/Debug/render_signal.sh ./output/iir_step_pulse.csv

Output

DspCore :: Generic IIR step-pulse response
Wrote signal data: ./output/iir_step_pulse.csv
PNG: .../render_signal.sh ./output/iir_step_pulse.csv

This is intentionally a time-only view: an FFT of a finite step pulse would describe the stimulus, not the IIR transfer function. The plot shows the normalized recurrence reaching its unity DC response and decaying after the falling edge.

#include "Shared.h"
#include <DspCore/Filters/SingleRate/IIR.h>
#include <stdio.h>
namespace DspCoreExamples { namespace Filters { namespace IIR {
bool Run(void (*print)(const char*)) {
DspCore::Filters::IIR<float> filter;
const float numerator[]={1,2,3,4},denominator[]={2,-1};
if(!filter.Configure(numerator,denominator)) return false;
const float impulse[]={1.f,0.f,0.f,0.f,0.f};
float output[5]={};
if(!filter.Process(impulse,output,5)) return false;
const float expected[]={0.5f,1.25f,2.125f,3.0625f,1.53125f};
bool ok=true;char text[100];
print("Generic IIR: 4 numerator, 2 denominator coefficients; a0=2 normalized internally");
for(unsigned i=0;i<5;++i) {
snprintf(text,sizeof(text),"Impulse[%u]=%d/32; expected=%d/32",i,static_cast<int>(output[i]*32),static_cast<int>(expected[i]*32));
print(text);ok=ok && output[i]==expected[i];
}
return ok;
}
