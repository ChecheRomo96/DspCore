Map physical and normalized coordinates through a linear segment.

Goal

Separate the coordinate system from the signal values: interpolate at a physical point, intentionally extrapolate a normalized point, and observe how a collapsed input axis selects its 0-side.

Run

./build/macos_arm64/bin/Debug/DspCore_Interpolation_Linear_d

Output

Physical x=15 in [10,20]: value=5; expected=5
Normalized t=2: extrapolated=14; expected=14
Collapsed axis selects value0: PASS
Overall: PASS

Extrapolation is a deliberate linear continuation. A collapsed physical axis maps to normalized zero, avoiding division by zero and selecting the value at that axis' 0 endpoint.

Implementation

#include "Shared.h"
#include <DspCore_Interpolation.h>
#include <stdio.h>
namespace DspCoreExamples { namespace Interpolation { namespace Linear {
bool Run(void (*print)(const char*)) {
const double value = DspCore::Interpolation::LinearAt(10.,20.,2.,8.,15.);
bool ok = value == 5.;
char message[96];
snprintf(message,sizeof(message),"Physical x=15 in [10,20]: value=%d; expected=5",static_cast<int>(value));
print(message);
const double extrapolated = DspCore::Interpolation::NormalizedLinear(2.,8.,2.);
snprintf(message,sizeof(message),"Normalized t=2: extrapolated=%d; expected=14",static_cast<int>(extrapolated));
print(message); ok = ok && extrapolated == 14.;
const double collapsed = DspCore::Interpolation::LinearAt(2.,2.,5.,9.,100.);
print(collapsed == 5. ? "Collapsed axis selects value0: PASS" : "Collapsed axis: FAIL");
return ok && collapsed == 5.;
}
}}}
