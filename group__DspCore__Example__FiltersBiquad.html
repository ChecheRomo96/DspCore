Visualize a 12 kHz low-pass/high-pass Biquad handoff.

Goal

Send a 1–200 kHz logarithmic sine sweep through complementary low-pass and high-pass Biquads at 480 kHz, with a 12 kHz cutoff and Q=0.7071.

Run and render

./build/macos_arm64/bin/Debug/DspCore_Filters_Biquad_d --output ./output
./build/macos_arm64/bin/Debug/render_signal.sh ./output/biquad_sweep.csv

Output

DspCore :: Biquad 12 kHz low-pass / high-pass sweep
Wrote signal data: ./output/biquad_sweep.csv
PNG: .../render_signal.sh ./output/biquad_sweep.csv

The upper plot is a short RMS envelope of the temporal sweep, so the handoff remains readable. The lower plot is the direct, coefficient-derived response on a logarithmic frequency axis. Its guides identify the 12 kHz crossover and -3 dB point.

#include "../../../Desktop/SignalCsv.h"
#include <DspCore_Filters.h>
#include <array>
#include <cmath>
#include <cstdio>
int main(int argc,char* argv[]) {
std::filesystem::path outputDirectory;
if(!DspCoreExamples::Desktop::ParseOutputDirectory(argc,argv,outputDirectory)) return 2;
constexpr size_t sampleCount=8192;
constexpr float sampleRate=48000.f;
constexpr float pi=3.14159265358979323846f;
std::array<float,sampleCount> input={};
std::array<float,sampleCount> lowPass={};
std::array<float,sampleCount> highPass={};
float phase=0.f;
for(size_t sample=0;sample<sampleCount;++sample) {
const float position=static_cast<float>(sample)/static_cast<float>(sampleCount-1);
const float frequency=100.f*std::pow(20000.f/100.f,position);
phase+=2.f*pi*frequency/sampleRate;
input[sample]=0.75f*std::sin(phase);
}
DspCore::Filters::IIR<float> lowPassFilter;
DspCore::Filters::IIR<float> highPassFilter;
if(!lowPassFilter.Configure(DspCore::Filters::Biquad::LowPass(sampleRate,12000.f,0.70710678f)) ||
!highPassFilter.Configure(DspCore::Filters::Biquad::HighPass(sampleRate,12000.f,0.70710678f)) ||
!lowPassFilter.Process(input.data(),lowPass.data(),lowPass.size()) ||
!highPassFilter.Process(input.data(),highPass.data(),highPass.size())) return 1;
const DspCoreExamples::Desktop::SignalColumn columns[]={
{"input",input.data()}, {"lowpass_12khz",lowPass.data()}, {"highpass_12khz",highPass.data()}};
std::filesystem::path outputPath;
if(!DspCoreExamples::Desktop::WriteSignalCsv(outputDirectory,"biquad_sweep.csv",
"Biquad low-pass and high-pass sweep","100 Hz to 20 kHz sine sweep; 12 kHz cutoff, Q=0.7071",
"time_and_spectrum",sampleRate,columns,3,sampleCount,outputPath)) {
std::fputs("Could not write CSV. Use --output <writable directory>.\n",stderr);
return 1;
}
std::puts("DspCore :: Biquad 12 kHz low-pass / high-pass sweep");
DspCoreExamples::Desktop::PrintPlotHint(outputPath,argv[0]);
return 0;
}

#include "Shared.h"
#include <DspCore_Filters.h>
#include <stdio.h>
namespace DspCoreExamples { namespace Filters { namespace Biquad {
bool Run(void (*print)(const char*)) {
auto coefficients=DspCore::Filters::Biquad::LowPass(48000.f,12000.f,0.70710678f);
DspCore::Filters::IIR<float> filter;
if(!filter.Configure(coefficients)) return false;
const float impulse[]={1.f}; float firstOutput[1]={};
if(!filter.Process(impulse,firstOutput,1)) return false;
char text[96];
bool ok=fabsf(firstOutput[0]-0.29289322f)<1e-5f;
snprintf(text,sizeof(text),"Low-pass first impulse sample=%d/1000; expected=293/1000",static_cast<int>(firstOutput[0]*1000+0.5f)); print(text);
const float constant[]={1.f,1.f,1.f,1.f,1.f,1.f,1.f,1.f,1.f,1.f,1.f,1.f,1.f,1.f,1.f,1.f};
float output[16]={};
for(unsigned i=0;i<8;++i) if(!filter.Process(constant,output,16)) return false;
ok=ok && fabsf(output[15]-1.f)<1e-5f;
print("Low-pass constant input=1 settles to 1");
coefficients=DspCore::Filters::Biquad::HighPass(48000.f,12000.f,0.70710678f);
if(!filter.Configure(coefficients)) return false;
for(unsigned i=0;i<8;++i) if(!filter.Process(constant,output,16)) return false;
print("High-pass constant input=1 settles to 0");
return ok && fabsf(output[15])<1e-5f;
}
