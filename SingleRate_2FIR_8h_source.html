183 bool processSample(const SampleType input, SampleType& output) noexcept {
184 if (coefficients_.empty()) {
185 output = input;

188
189 const StateType stateInput = static_cast<StateType>(input);
190 if (!DspCore::Detail::Signal::Finite(stateInput)) {

193
194 StateType sum = static_cast<StateType>(coefficients_[0]) * stateInput;
195 cpstd::size_t historyIndex = next_;
196 for (cpstd::size_t tap = 1U; tap < coefficients_.size(); ++tap) {
197 historyIndex = historyIndex == 0U ? coefficients_.size() - 1U : historyIndex - 1U;
198 sum += static_cast<StateType>(coefficients_[tap]) * history_[historyIndex];
199 }
200 if (!DspCore::Detail::Signal::Finite(sum)) {
201 return false;
202 }
203 SampleType converted{};

205 return false;
206 }
207
208 history_[next_] = stateInput;
209 next_ = next_ == coefficients_.size() - 1U ? 0U : next_ + 1U;
210 output = converted;
211 return true;
