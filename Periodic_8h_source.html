11 !Generators::Detail::Finite(phase)) return false;
12 const Real start=Generators::Detail::Wrap(phase);
13 for (size_t i=0;i<count;++i) {
14 Real p=Generators::Detail::Wrap(start+static_cast<Real>(i)/static_cast<Real>(count));
15 Real sample=Real(0);

18 case Wave::Saw: sample=Real(2)*Generators::Detail::Wrap(p+Real(0.5))-Real(1); break;
19 case Wave::Square: sample=p < Real(0.5) ? Real(1) : Real(-1); break;
20 case Wave::Triangle:
21 p=Generators::Detail::Wrap(p+Real(0.75));
22 sample=Real(2)*Generators::Detail::Abs(Real(2)*p-Real(1))-Real(1); break;
23 }
24 buffer[i]=offset+amplitude*sample;
25 }
26 return true;
27}
