Compare 10- and 100-tap moving-average low-pass filters.

Compare 10- and 100-tap moving-average low-pass filters.

Goal

Process the same 300 Hz + 6 kHz signal with two runtime-sized FIR filters. The longer average suppresses more high-frequency content, while its linear-phase delay increases from 0.094 ms to 1.031 ms at 48 kHz.

Run and render

./build/macos_arm64/bin/Debug/DspCore_Filters_FIR_d --output ./output
./build/macos_arm64/bin/Debug/render_signal.sh ./output/fir_moving_average.csv

Desktop data generator

#include "../../../Desktop/SignalCsv.h"
#include <DspCore_Filters.h>
#include <array>
#include <cmath>
#include <cstdio>
int main(int argc,char* argv[]) {
std::filesystem::path outputDirectory;
if(!DspCoreExamples::Desktop::ParseOutputDirectory(argc,argv,outputDirectory)) return 2;
constexpr size_t sampleCount=4096;
constexpr float sampleRate=48000.f;
constexpr float pi=3.14159265358979323846f;
std::array<float,33> taps={};
for(float& tap:taps) tap=1.f/static_cast<float>(taps.size());
std::array<float,sampleCount> input={};
std::array<float,sampleCount> output={};
for(size_t sample=0;sample<sampleCount;++sample) {
const float time=static_cast<float>(sample)/sampleRate;
input[sample]=0.6f*std::sin(2.f*pi*300.f*time)+0.3f*std::sin(2.f*pi*6000.f*time);
}
DspCore::Filters::FIR<float> filter;
if(!filter.Configure(taps.data(),taps.size()) ||
!filter.Process(input.data(),output.data(),output.size())) return 1;
const DspCoreExamples::Desktop::SignalColumn columns[]={
{"input",input.data()}, {"moving_average_33",output.data()}};
std::filesystem::path outputPath;
if(!DspCoreExamples::Desktop::WriteSignalCsv(outputDirectory,"fir_moving_average.csv",
"FIR moving-average low-pass","300 Hz + 6 kHz input; 33-tap moving average",
"time_and_spectrum",sampleRate,
columns,2,sampleCount,outputPath)) {
std::fputs("Could not write CSV. Use --output <writable directory>.\n",stderr);
return 1;
}
std::puts("DspCore :: FIR moving-average low-pass");
DspCoreExamples::Desktop::PrintPlotHint(outputPath,argv[0]);
return 0;
}

Portable FIR API use

#include "Shared.h"
#include <DspCore_Filters.h>
#include <stdio.h>
namespace DspCoreExamples { namespace Filters { namespace FIR {
bool Run(void (*print)(const char*)) {
DspCore::Filters::FIR<float> filter;
const float taps[]={0.25f,0.25f,0.25f,0.25f};
if(!filter.Configure(taps,4)) return false;
const float impulse[]={1.f,0.f,0.f,0.f,0.f};
float output[5]={};
if(!filter.Process(impulse,output,5)) return false;
char text[96]; bool ok=true;
for(unsigned i=0;i<5;++i) {
const int expected=i<4 ? 25 : 0;
snprintf(text,sizeof(text),"Impulse[%u]=%d%%; expected=%d%%",i,static_cast<int>(output[i]*100),expected); print(text);
ok=ok && output[i]==(i<4 ? 0.25f : 0.f);
}
filter.Reset();
const float constant[]={1.f,1.f,1.f,1.f,1.f,1.f,1.f,1.f};
float settled[8]={};
if(!filter.Process(constant,settled,8)) return false;
print("Constant input=1 settles exactly to 1 (unity DC gain)");
return ok && settled[7]==1;
}
}}}
