79 bool Transform(const Processing::SampleBlock<const Complex> input,
80 const Processing::SampleBlock<Complex> output,const bool inverse) noexcept {
81 if(!Valid(input,output)) return false;
82 for(cpstd::size_t index=0U;index<size_;++index) {
83 if(!Finite(input.data[index])) return false;
84 workspace_[index]=input.data[index];
85 }
86 BitReverse();
87 const Real direction=inverse ? Real(1) : Real(-1);
88 for(cpstd::size_t length=2U;;length<<=1U) {
89 const Real angle=direction*Real(2)*Detail::Signal::Pi<Real>()/static_cast<Real>(length);
90 Complex step;
91 step.real=Detail::Signal::Cos(angle);
92 step.imag=Detail::Signal::Sin(angle);
93 const cpstd::size_t half=length/2U;
94 for(cpstd::size_t start=0U;start<size_;start+=length) {
95 Complex twiddle;
96 twiddle.real=Real(1); twiddle.imag=Real(0);
97 for(cpstd::size_t offset=0U;offset<half;++offset) {
98 const Complex upper=workspace_[start+offset];
99 const Complex lower=workspace_[start+offset+half];
100 Complex product;
101 product.real=lower.real*twiddle.real-lower.imag*twiddle.imag;
102 product.imag=lower.real*twiddle.imag+lower.imag*twiddle.real;
103 workspace_[start+offset].real=upper.real+product.real;
104 workspace_[start+offset].imag=upper.imag+product.imag;
105 workspace_[start+offset+half].real=upper.real-product.real;
106 workspace_[start+offset+half].imag=upper.imag-product.imag;
107 if(!Finite(workspace_[start+offset]) || !Finite(workspace_[start+offset+half])) return false;
108 const Real nextReal=twiddle.real*step.real-twiddle.imag*step.imag;
109 twiddle.imag=twiddle.real*step.imag+twiddle.imag*step.real;
110 twiddle.real=nextReal;
111 if(!Finite(twiddle)) return false;
112 }
113 }
114 if(length==size_) break;
115 }
116 const Real scale=inverse ? Real(1)/static_cast<Real>(size_) : Real(1);
117 for(cpstd::size_t index=0U;index<size_;++index) {
118 workspace_[index].real*=scale;
119 workspace_[index].imag*=scale;
120 if(!Finite(workspace_[index])) return false;
121 }
122 for(cpstd::size_t index=0U;index<size_;++index) output.data[index]=workspace_[index];
123 return true;
124 }
125
126 void BitReverse() noexcept {
127 cpstd::size_t reversed=0U;
128 for(cpstd::size_t index=1U;index<size_;++index) {
129 cpstd::size_t bit=size_>>1U;
130 while((reversed&bit)!=0U) { reversed^=bit; bit>>=1U; }
131 reversed^=bit;
132 if(index<reversed) {
133 const Complex temporary=workspace_[index];
134 workspace_[index]=workspace_[reversed];
135 workspace_[reversed]=temporary;
136 }
137 }
138 }

140 cpstd::vector<Complex> workspace_;
141 cpstd::size_t size_{};
