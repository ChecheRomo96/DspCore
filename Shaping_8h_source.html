8bool Sigmoid(Real* buffer, size_t count, Real drive, Real scale, bool atanCurve, bool half, bool normalized) noexcept {
9 if (!Generators::Detail::Buffer(buffer,count) || !Generators::Detail::Finite(drive) || drive<=Real(0) ||
10 !Generators::Detail::Finite(scale) || scale<Real(0)) return false;
11 const Real denominator = atanCurve ? Generators::Detail::Atan(drive) : Generators::Detail::Tanh(drive);
12 for(size_t i=0;i<count;++i) {
13 const Real t=count>1 ? static_cast<Real>(i)/static_cast<Real>(count-1) : (half ? Real(0) : Real(0.5));
14 const Real x=half ? t : Real(2)*t-Real(1);
15 Real y=atanCurve ? Generators::Detail::Atan(drive*x) : Generators::Detail::Tanh(drive*x);
16 if (normalized) {
17 y=drive <= Real(8)*cpstd::numeric_limits<Real>::epsilon() ? x : y/denominator;
18 const Real magnitude=Generators::Detail::Unit(Generators::Detail::Abs(y));
19 y=x<Real(0) ? -magnitude : magnitude;
20 if (count>1 && i==0) y=half ? Real(0) : Real(-1);
21 if (count>1 && i==count-1) y=Real(1);
22 }
23 // Raw atan is bounded by pi/2 rather than one; validate its scale below.
24 buffer[i]=y*scale;
25 }
26 return true;
27}
