20template<typename Real>
21Real Linear(Real a, Real b, Real t) noexcept {
22 static_assert(cpstd::is_same<Real, float>::value || cpstd::is_same<Real, double>::value,
23 "Interpolation arithmetic must explicitly use float or double");
24 if (t == Real(0)) return a;
25 if (t == Real(1)) return b;
26 // Weighted form avoids overflowing b-a for opposite-sign finite endpoints.
27 if ((a <= Real(0) && b >= Real(0)) || (a >= Real(0) && b <= Real(0)))
28 return (Real(1) - t) * a + t * b;
29 return a + t * (b - a);
30}

49template<typename Real>
50bool TryUnitCoordinate(Real x, Real low, Real high, Real& t) noexcept {
51 if (!Detail::Finite(x) || !Detail::Finite(low) || !Detail::Finite(high) || !(low < high))
52 return false;
53 const Real span = high - low;
54 if (!Detail::Finite(span)) return false;
55 const Real result = (x - low) / span;
56 if (!Detail::Finite(result)) return false;
57 t = result;
58 return true;
