24template<typename Real, typename T>
25Real SampleValue(T value) noexcept {
26 const long double wide = static_cast<long double>(value);
27 const long double limit = static_cast<long double>(cpstd::numeric_limits<Real>::max());
28 if (wide != wide) return cpstd::numeric_limits<Real>::quiet_NaN();
29 if (wide > limit) return cpstd::numeric_limits<Real>::infinity();
30 if (wide < -limit) return -cpstd::numeric_limits<Real>::infinity();
31 return static_cast<Real>(wide);
32}

34template<typename Real>
35bool SamplingAxis(size_t count, Real coordinate, Boundary boundary,
36 size_t& first, size_t& second, Real& fraction) noexcept {
37 if (count == 0 || !FiniteCoordinate(coordinate)) return false;
38 if (boundary != Boundary::Clamp && boundary != Boundary::Wrap && boundary != Boundary::Reject)
39 return false;
40 if (boundary == Boundary::Reject && (coordinate < Real(0) || coordinate > Real(1))) return false;
41 if (count == 1) { first = second = 0; fraction = Real(0); return true; }
42 Real normalized = coordinate;
43 const bool wrap = boundary == Boundary::Wrap;
44 if (wrap) {
45 normalized = static_cast<Real>(fmod(static_cast<double>(coordinate), 1.0));
46 if (normalized < Real(0)) normalized += Real(1);
47 // Negative sub-ulp offsets can round to one after addition.
48 if (normalized >= Real(1)) normalized = Real(0);
49 } else {
50 if (normalized <= Real(0)) { first = second = 0; fraction = Real(0); return true; }
51 if (normalized >= Real(1)) { first = second = count - 1; fraction = Real(0); return true; }
52 }
53 // Require the axis extent to fit the calculation type's exact-integer range.
54 // This also prevents rounding an index beyond size_t before its cast.
55 const Real exactLimit = static_cast<Real>(ldexp(1.0, cpstd::numeric_limits<Real>::digits));
56 const Real extent = static_cast<Real>(count);
57 if (extent > exactLimit || extent >= static_cast<Real>(static_cast<size_t>(-1))) return false;
58 if (static_cast<size_t>(extent) != count) return false;
59 const Real scale = static_cast<Real>(wrap ? count : count - 1);
60 const Real position = normalized * scale;
61 if (position >= scale) {
62 first = count - 1;
63 second = wrap ? 0 : first;
64 fraction = wrap ? Real(1) : Real(0);
65 return true;
66 }
67 first = static_cast<size_t>(position);
68 second = first + 1;
69 if (second >= count) second = wrap ? 0 : first;
70 fraction = position - static_cast<Real>(first);
71 return true;
72}
